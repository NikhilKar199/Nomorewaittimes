Sort Theme Park Rides

Enter ride names separated by commas and click Sort.

from js import document, console def sort_rides(event=None): try: input_el = document.getElementById("ride-input") output_el = document.getElementById("sorted-rides") if not input_el or not output_el: return rides = input_el.value or "" ride_list = [r.strip() for r in rides.split(",") if r.strip()] ride_list.sort() output_el.innerText = ", ".join(ride_list) except Exception as e: try: console.error(f"sort_rides error: {e}") except Exception: pass btn = document.getElementById("sort-rides-btn") if btn: btn.addEventListener("click", sort_rides)

Efficient Ride Layout

Paste ride data (CSV: name,x,y,duration_min,wait_min,priority[1-5]). Higher priority means more important.

from math import sqrt def parse_csv(text): lines = [ln.strip() for ln in text.split('\n') if ln.strip()] rides = [] for ln in lines: parts = [p.strip() for p in ln.split(',')] if len(parts) < 6: # Try to be forgiving: pad with defaults while len(parts) < 6: parts.append('0') name = parts[0] try: x = float(parts[1]); y = float(parts[2]) duration = float(parts[3]); wait = float(parts[4]) priority = float(parts[5]) if parts[5] else 3.0 except Exception: # skip malformed lines continue rides.append({ 'name': name, 'x': x, 'y': y, 'duration': duration, 'wait': wait, 'priority': priority, }) return rides def distance(a, b): return sqrt((a['x']-b['x'])**2 + (a['y']-b['y'])**2) def score(curr, nxt, w_dist=1.0, w_wait=0.15, w_priority=0.5): # lower score is better d = distance(curr, nxt) # Subtract priority so higher priority reduces the cost return w_dist*d + w_wait*max(nxt['wait'], 0) - w_priority*max(nxt['priority'], 0) def compute_route(rides): if not rides: return [] start = {'name': 'Start', 'x': 0.0, 'y': 0.0, 'duration': 0.0, 'wait': 0.0, 'priority': 0.0} remaining = rides[:] route = [] current = start while remaining: nxt = min(remaining, key=lambda r: score(current, r)) route.append(nxt) remaining.remove(nxt) current = nxt return route def on_compute(event=None): raw = document.getElementById('ride-data').value or '' exclude = {n.strip().lower() for n in (document.getElementById('exclude-rides').value or '').split(',') if n.strip()} rides = [r for r in parse_csv(raw) if r['name'].lower() not in exclude] route = compute_route(rides) lines = [] total_walk = 0.0 total_wait = 0.0 prev = {'name': 'Start', 'x': 0.0, 'y': 0.0, 'duration': 0.0, 'wait': 0.0, 'priority': 0.0} for i, r in enumerate(route, 1): d = distance(prev, r) total_walk += d total_wait += max(r['wait'], 0) lines.append(f"{i}. {r['name']} [walk≈{d:.1f}, wait≈{r['wait']:.0f}m, priority={r['priority']:.0f}]") prev = r if not lines: lines = ["No rides parsed. Paste CSV data and try again."] summary = f"Total estimated walk distance≈{total_walk:.1f} | Total wait≈{total_wait:.0f}m"
